# ============================================================================
# AUDIO/SONIC BRAND IDENTITY
# ============================================================================

audio_identity:
  
  core_sound:
    essence: "Acoustic-driven pop ballads and mid-tempo songs that feel like a journal entry set to music - intimate, encouraging, and perfectly pitched for pre-teen hearts"
    production_quality: "Disney Channel meets Spotify-worthy worship - polished but not overproduced, warm and inviting"
    reference_artists:
      - "Early Taylor Swift - Fearless era acoustic storytelling and relatable lyrics"
      - "Olivia Rodrigo ballads - emotional honesty and teen perspective"
      - "Kari Jobe - gentle worship vocals and piano-driven arrangements"
      - "Lauren Daigle acoustic - accessible contemporary Christian sound"
      - "Sabrina Carpenter - bubblegum pop energy without being too secular"
    
  sonic_characteristics:
    texture: "Warm and close, like a friend singing in your bedroom"
    space: "Intimate with gentle reverb, never cavernous or distant"
    dynamics: "Gentle verses building to confident choruses, not aggressive"
    energy: "Ranges from quiet morning devotion to upbeat youth group singalong"
  
  instrumentation:
    primary:
      - "Acoustic guitar (fingerpicking and gentle strumming - foundation)"
      - "Piano (emotional center, storytelling partner)"
      - "Light percussion (shakers, soft snare, gentle kick - not full kit)"
      - "Bass (subtle, supportive, never overpowering)"
    
    atmospheric:
      - "Light synth pads (dreamy, not EDM - think bedroom producer vibes)"
      - "Ukulele (for playful, joyful Psalms)"
      - "String sections (on special moments, not everywhere)"
      - "Music box tinkles (for gentle intros/outros)"
      
    avoid:
      - "Heavy electric guitars or distortion"
      - "Complex drum programming or trap beats"
      - "Aggressive synths or EDM drops"
      - "Adult contemporary production that sounds too polished"
      - "Childish sing-song melodies that talk down"
  
  vocal_style:
    lead_vocal:
      type: "Young female voice (sounds 16-18, relatable to 12-year-olds)"
      quality: "Clear, sweet, emotionally honest - not technically perfect but authentic"
      technique: "Conversational verses, confident choruses, never belting or showing off"
      range: "Singable in car/bedroom - stays in comfortable range (no Whitney Houston runs)"
      
    backing_vocals:
      type: "Gentle harmonies, occasional 'ooh' layers, youth group feel"
      style: "Supportive not dominant, like friends singing together"
      
  tempo_feel:
    ballad_range: "70-85 BPM (quiet time with God, bedtime prayers)"
    mid_tempo: "90-105 BPM (getting ready energy, walking to school confidence)"
    uptempo: "110-125 BPM (youth group joy, summer camp worship)"
    
  structure:
    typical_form: "Intro - Verse 1 - Chorus - Verse 2 - Chorus - Bridge - Final Chorus - Outro"
    bridge_importance: "Key moment of truth or encouragement - where the psalm speaks directly to listener"
    intro_style: "Acoustic guitar hook or piano melody (4-8 bars, inviting)"
    outro_style: "Gentle fade or repeated comfort phrase (like tucking into bed)"
